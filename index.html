// ========================================================================= // HỆ THỐNG QUẢN LÝ TSTC - TOÀN BỘ CODE GOOGLE APPS SCRIPT HOÀN CHỈNH // ========================================================================= function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. LẤY DỮ LIỆU BAN ĐẦU (CHỈ LẤY CÁC MÃ 'KIỂM KHO' ĐỂ CHỐNG TRÙNG CHUẨN XÁC) if (action == "getData") { var sheetDS = ss.getSheetByName("DanhSach"); var sheetQR = ss.getSheetByName("QuetQR"); var dataDS = sheetDS.getDataRange().getValues(); var listDS = []; for (var i = 1; i < dataDS.length; i++) { if(dataDS[i][0]) { listDS.push({ soHD: dataDS[i][0].toString().trim(), hoten: dataDS[i][1] ? dataDS[i][1].toString() : "", sluong: Number(dataDS[i][2]) || 0, trangThai: dataDS[i][4] ? dataDS[i][4].toString().trim() : "" }); } } // CHỈ LẤY CÁC MÃ ĐÃ QUÉT 'KIỂM KHO' ĐỂ CHỐNG TRÙNG DÀNH RIÊNG CHO CHẾ ĐỘ KIỂM KHO var dataQR = sheetQR.getDataRange().getValues(); var listQR = []; for (var j = 1; j < dataQR.length; j++) { var thaoTac = dataQR[j][5] ? dataQR[j][5].toString().toUpperCase().trim() : ""; if(dataQR[j][0] && thaoTac === "KIỂM KHO") { listQR.push(dataQR[j][0].toString().trim()); } } return ContentService.createTextOutput(JSON.stringify({listDS: listDS, listQR: listQR})).setMimeType(ContentService.MimeType.TEXT_PLAIN); } // 2. XUẤT BÁO CÁO KIỂM KHO SANG SHEET 'KetQua' if (action == "tinhKetQua") { var sheetDS = ss.getSheetByName("DanhSach"); var sheetQR = ss.getSheetByName("QuetQR"); var sheetKQ = ss.getSheetByName("KetQua"); var lastRowKQ = sheetKQ.getLastRow(); if (lastRowKQ > 1) { sheetKQ.getRange(2, 1, lastRowKQ - 1, 4).clearContent(); } var dataDS = sheetDS.getDataRange().getValues(); var dataQR = sheetQR.getDataRange().getValues(); var danhMucGoc = {}; var khoHeThong = {}; for (var i = 1; i < dataDS.length; i++) { var soHD = dataDS[i][0].toString().trim(); if (soHD) { danhMucGoc[soHD] = dataDS[i][1] || "Không rõ"; var tt = dataDS[i][4] ? dataDS[i][4].toString().trim() : ""; if (tt === "Trong kho") { khoHeThong[soHD] = { hoten: dataDS[i][1], sluong: Number(dataDS[i][2]) || 0, quetKiemTra: 0 }; } } } var khoThucTe = {}; for (var j = 1; j < dataQR.length; j++) { var thaoTac = dataQR[j][5] ? dataQR[j][5].toString().toUpperCase().trim() : ""; if (thaoTac === "KIỂM KHO") { var soHD_QR = dataQR[j][3].toString().trim(); var sluong_QR = Number(dataQR[j][4]) || 0; if (soHD_QR) { khoThucTe[soHD_QR] = (khoThucTe[soHD_QR] || 0) + sluong_QR; if (khoHeThong[soHD_QR]) { khoHeThong[soHD_QR].quetKiemTra += sluong_QR; } } } } var rowsKQ = []; for (var soHD in khoHeThong) { if (khoHeThong[soHD].quetKiemTra === 0) { rowsKQ.push([soHD, khoHeThong[soHD].hoten, "Thiếu thực tế", "Hệ thống báo 'Trong kho' nhưng kiểm kê không thấy"]); } } for (var soHD_QR in khoThucTe) { if (!khoHeThong[soHD_QR]) { var tenKhach = danhMucGoc[soHD_QR] || "Không rõ"; var ghiChu = !danhMucGoc[soHD_QR] ? "Bao này nằm ngoài danh sách tồn kho hiện tại" : "Bao có trong danh sách nhưng trạng thái trên hệ thống không phải 'Trong kho'"; rowsKQ.push([soHD_QR, tenKhach, "Thừa hệ thống", ghiChu]); } } if (rowsKQ.length > 0) { sheetKQ.getRange(2, 1, rowsKQ.length, 4).setValues(rowsKQ); } return ContentService.createTextOutput(JSON.stringify({status: "success", count: rowsKQ.length})).setMimeType(ContentService.MimeType.TEXT_PLAIN); } // 3. XUẤT BÁO CÁO NHẬP - XUẤT - TỒN DỰA TRÊN MỐC NGÀY TẠI I1 VÀ K1 if (action == "chayBaoCaoDinhKy") { var sheetBC = ss.getSheetByName("BaoCao"); var sheetQR = ss.getSheetByName("QuetQR"); var sheetDS = ss.getSheetByName("DanhSach"); var tuNgayStr = sheetBC.getRange("I1").getDisplayValue().toString().trim(); var denNgayStr = sheetBC.getRange("K1").getDisplayValue().toString().trim(); if(!tuNgayStr || !denNgayStr) { return ContentService.createTextOutput(JSON.stringify({status: "error", msg: "Chưa điền ngày tại I1 hoặc K1"})).setMimeType(ContentService.MimeType.TEXT_PLAIN); } function dateToNumber(str) { var datePart = str.split(" ")[0].includes("/") ? str.split(" ")[0] : str.split(" ")[1]; var p = datePart.replace(",", "").split("/"); var d = p[0].padStart(2, '0'); var m = p[1].padStart(2, '0'); var y = p[2]; if (y.length < 4) y = "20" + y; return parseInt(y + m + d, 10); } var tuNgayNum = dateToNumber(tuNgayStr); var denNgayNum = dateToNumber(denNgayStr); var lastRow = sheetBC.getLastRow(); if (lastRow > 1) { sheetBC.getRange(2, 1, lastRow - 1, 6).clearContent(); sheetBC.getRange(2, 1, lastRow - 1, 6).setFontWeight("normal"); } var dataDS = sheetDS.getDataRange().getValues(); var rangeQR = sheetQR.getDataRange(); var dataQRValues = rangeQR.getValues(); var dataQRDisplays = rangeQR.getDisplayValues(); var mapBaoCao = {}; for (var i = 1; i < dataDS.length; i++) { var soHD = dataDS[i][0].toString().trim(); if(soHD && !mapBaoCao[soHD]) { mapBaoCao[soHD] = { hoten: dataDS[i][1] || "Không rõ", dauKy: 0, nhapKho: 0, xuatKho: 0 }; } } for (var j = 1; j < dataQRValues.length; j++) { var ngayQuetStr = dataQRDisplays[j][1].toString().trim(); var soHD_QR = dataQRValues[j][3].toString().trim(); var sluong_QR = Number(dataQRValues[j][4]) || 0; var thaoTac = dataQRValues[j][5] ? dataQRValues[j][5].toString().toUpperCase().trim() : ""; if (!soHD_QR || !ngayQuetStr) continue; if (thaoTac !== "NHẬP KHO" && thaoTac !== "XUẤT KHO") continue; if (!mapBaoCao[soHD_QR]) { mapBaoCao[soHD_QR] = { hoten: "Mã phát sinh ngoài", dauKy: 0, nhapKho: 0, xuatKho: 0 }; } var ngayQuetNum = dateToNumber(ngayQuetStr); if (ngayQuetNum < tuNgayNum) { if (thaoTac === "NHẬP KHO") mapBaoCao[soHD_QR].dauKy += sluong_QR; if (thaoTac === "XUẤT KHO") mapBaoCao[soHD_QR].dauKy -= sluong_QR; } else if (ngayQuetNum >= tuNgayNum && ngayQuetNum <= denNgayNum) { if (thaoTac === "NHẬP KHO") mapBaoCao[soHD_QR].nhapKho += sluong_QR; if (thaoTac === "XUẤT KHO") mapBaoCao[soHD_QR].xuatKho += sluong_QR; } } var rowsBC = []; var tongDauKy = 0, tongNhap = 0, tongXuat = 0, tongCuoiKy = 0; for (var soHD in mapBaoCao) { var item = mapBaoCao[soHD]; var cuoiKy = item.dauKy + item.nhapKho - item.xuatKho; if (item.dauKy === 0 && item.nhapKho === 0 && item.xuatKho === 0 && cuoiKy === 0) continue; tongDauKy += item.dauKy; tongNhap += item.nhapKho; tongXuat += item.xuatKho; tongCuoiKy += cuoiKy; rowsBC.push([soHD, item.hoten, item.dauKy, item.nhapKho, item.xuatKho, cuoiKy]); } rowsBC.push(["TỔNG CỘNG", "", tongDauKy, tongNhap, tongXuat, tongCuoiKy]); if (rowsBC.length > 0) { sheetBC.getRange(2, 1, rowsBC.length, 6).setValues(rowsBC); sheetBC.getRange(sheetBC.getLastRow(), 1, 1, 6).setFontWeight("bold"); } return ContentService.createTextOutput(JSON.stringify({status: "success", count: rowsBC.length})).setMimeType(ContentService.MimeType.TEXT_PLAIN); } } // 4. LƯU DỮ LIỆU KHI QUÉT TỪ ĐIỆN THOẠI function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetDS = ss.getSheetByName("DanhSach"); var sheetQR = ss.getSheetByName("QuetQR"); var p = e.parameter; var maQuet = p.maQuet || ""; var thoiGian = p.thoiGian || new Date().toLocaleString("vi-VN"); var email = p.email || "Thiết_Bị_Quét"; var soHD = p.soHD || ""; var sluong = Number(p.sluong) || 1; var loaiThaoTac = p.loaiThaoTac || ""; var hoten = p.hoten || ""; var tstc = p.tstc || ""; sheetQR.appendRow([maQuet, thoiGian, email, soHD, sluong, loaiThaoTac]); var thaoTacXuly = loaiThaoTac.toString().toUpperCase().trim(); var dataDS = sheetDS.getDataRange().getValues(); if (thaoTacXuly === "NHẬP KHO") { sheetDS.appendRow([soHD, hoten, sluong, tstc, "Trong kho", thoiGian, maQuet, "", ""]); } else if (thaoTacXuly === "XUẤT KHO") { for (var i = 1; i < dataDS.length; i++) { if (dataDS[i][0].toString().trim() === soHD.toString().trim() && dataDS[i][4].toString().trim() === "Trong kho") { sheetDS.getRange(i + 1, 5).setValue("Đã xuất"); sheetDS.getRange(i + 1, 6).setValue(thoiGian); break; } } } return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.TEXT_PLAIN); } // ========================================================================= // 5. HÀM TỰ ĐỘNG SAO LƯU FILE SANG GOOGLE DRIVE LÚC 15H45 HÀNG NGÀY // ========================================================================= function tuDongSaoLuuFile() { // Thay ID thư mục Drive của bạn tại đây var THU_MUC_BACKUP_ID = "1Gj2Kf6ALtoFJ30IO7GgPbs5dPQlhAkP9"; var ss = SpreadsheetApp.getActiveSpreadsheet(); var fileGoc = DriveApp.getFileById(ss.getId()); var thuMucLuu = DriveApp.getFolderById(THU_MUC_BACKUP_ID); var chuoiThoiGian = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyyMMdd_HHmmss"); var tenFileMoi = fileGoc.getName() + "_" + chuoiThoiGian; fileGoc.makeCopy(tenFileMoi, thuMucLuu); Logger.log("Đã sao lưu file thành công: " + tenFileMoi); } function thietLapThoiGianSaoLuu() { var triggers = ScriptApp.getProjectTriggers(); for (var i = 0; i < triggers.length; i++) { if (triggers[i].getHandlerFunction() === "tuDongSaoLuuFile") { ScriptApp.deleteTrigger(triggers[i]); } } ScriptApp.newTrigger("tuDongSaoLuuFile") .timeBased() .everyDays(1) .atHour(15) .create(); Logger.log("Đã thiết lập bộ hẹn giờ thành công!"); }
